cmake_minimum_required(VERSION 3.18)
project(sdb VERSION 0.3.0 LANGUAGES C CXX ASM)

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux" OR NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|amd64|AMD64)$")
    message(FATAL_ERROR "SDB requires Linux x86-64. On Apple Silicon use the project's x86-64 Colima VM.")
endif()

option(SDB_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer for SDB" OFF)
find_path(CAPSTONE_INCLUDE_DIR capstone/capstone.h REQUIRED)
find_library(CAPSTONE_LIBRARY capstone REQUIRED)

add_library(sdb_core
    src/tracee.cpp src/signal_monitor.cpp src/memory.cpp src/elf_image.cpp src/breakpoint.cpp
    src/watchpoint.cpp src/register_file.cpp src/backtrace.cpp src/syscall_decoder.cpp
    src/disassembler.cpp src/debugger.cpp src/command_parser.cpp)
target_compile_features(sdb_core PUBLIC cxx_std_17)
target_include_directories(sdb_core PUBLIC include "${CAPSTONE_INCLUDE_DIR}")
target_compile_options(sdb_core PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(sdb_core PUBLIC "${CAPSTONE_LIBRARY}")

add_executable(sdb src/main.cpp)
target_link_libraries(sdb PRIVATE sdb_core)
target_compile_options(sdb PRIVATE -Wall -Wextra -Wpedantic)
if(SDB_SANITIZERS)
    foreach(target sdb_core sdb)
        target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        target_link_options(${target} PUBLIC -fsanitize=address,undefined)
    endforeach()
endif()

include(CTest)
if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    find_package(Threads REQUIRED)
    foreach(unit inspection task_control detach)
        add_executable(${unit}_test tests/${unit}_test.cpp)
        target_link_libraries(${unit}_test PRIVATE sdb_core Threads::Threads)
        target_compile_options(${unit}_test PRIVATE -Wall -Wextra -Wpedantic
            -fno-omit-frame-pointer -fno-optimize-sibling-calls)
        if(SDB_SANITIZERS)
            target_compile_options(${unit}_test PRIVATE -fsanitize=address,undefined)
        endif()
        if(unit STREQUAL "detach")
            target_link_options(${unit}_test PRIVATE -Wl,--wrap=ptrace)
        elseif(unit STREQUAL "task_control")
            target_link_options(${unit}_test PRIVATE -Wl,--wrap=send)
        endif()
        add_test(NAME ${unit} COMMAND ${unit}_test)
        set_tests_properties(${unit} PROPERTIES TIMEOUT 90 SKIP_RETURN_CODE 77 LABELS unit)
    endforeach()
    add_executable(tracee_memory_test tests/tracee_memory_test.cpp)
    target_link_libraries(tracee_memory_test PRIVATE sdb_core)
    target_compile_options(tracee_memory_test PRIVATE -Wall -Wextra -Wpedantic)
    target_link_options(tracee_memory_test PRIVATE -Wl,--wrap=pread,--wrap=pwrite,--wrap=open)
    if(SDB_SANITIZERS)
        target_compile_options(tracee_memory_test PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    endif()
    add_test(NAME tracee_memory COMMAND tracee_memory_test)
    set_tests_properties(tracee_memory PROPERTIES TIMEOUT 60 SKIP_RETURN_CODE 77)
    foreach(mode pie exec)
        add_executable(elf_image_test_${mode} tests/elf_image_test.cpp)
        target_link_libraries(elf_image_test_${mode} PRIVATE sdb_core ${CMAKE_DL_LIBS})
        target_compile_options(elf_image_test_${mode} PRIVATE -Wall -Wextra -Wpedantic)
        if(mode STREQUAL "pie")
            target_compile_options(elf_image_test_${mode} PRIVATE -fPIE)
            target_link_options(elf_image_test_${mode} PRIVATE -pie)
        else()
            target_compile_options(elf_image_test_${mode} PRIVATE -fno-pie)
            target_link_options(elf_image_test_${mode} PRIVATE -no-pie)
        endif()
        if(SDB_SANITIZERS)
            target_compile_options(elf_image_test_${mode} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
        endif()
        add_test(NAME elf_image_${mode} COMMAND elf_image_test_${mode})
        set_tests_properties(elf_image_${mode} PROPERTIES TIMEOUT 30)
    endforeach()
    add_test(NAME integration COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/integration.py" "$<TARGET_FILE:sdb>")
    set_tests_properties(integration PROPERTIES TIMEOUT 300 SKIP_RETURN_CODE 77 LABELS integration)
    add_test(NAME extensions COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/extensions.py" "$<TARGET_FILE:sdb>")
    set_tests_properties(extensions PROPERTIES TIMEOUT 600 SKIP_RETURN_CODE 77 LABELS integration)
    foreach(suite elf_process signals interview_demo)
        add_test(NAME ${suite} COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/${suite}.py" "$<TARGET_FILE:sdb>")
        set_tests_properties(${suite} PROPERTIES TIMEOUT 300 SKIP_RETURN_CODE 77 LABELS integration)
    endforeach()
endif()
